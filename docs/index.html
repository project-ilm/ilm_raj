पूर्णाङ्क द्विआधारी_खोज(पूर्णाङ्क मूल्यावली[], पूर्णाङ्क गिनती, पूर्णाङ्क कुंजी) {
  पूर्णाङ्क निचला = 0, ऊपरी = गिनती - 1;
  यावत् (निचला <= ऊपरी) {
    पूर्णाङ्क मध्य = निचला + (ऊपरी - निचला) / 2;
    यदि (मूल्यावली[मध्य] == कुंजी) प्रत्यागम मध्य;
    यदि (मूल्यावली[मध्य] < कुंजी) निचला = मध्य + 1; अन्यथा ऊपरी = मध्य - 1;
  }
  प्रत्यागम -1;
}
पूर्णाङ्क मुख्य(शून्य) {
  पूर्णाङ्क मूल्यावली[] = {2, 5, 8, 12, 16, 23, 38, 56, 72, 91};
  printf("%d %d\n", द्विआधारी_खोज(मूल्यावली, 10, 23), द्विआधारी_खोज(मूल्यावली, 10, 7));
  प्रत्यागम 0;
}
